// first time pushing code to github rules

# 1. Go to project

cd ~/my-javaScript-Project

# 2. Initialize Git

git init

# 3. Use main branch

git branch -M main

# 4. Check files

git status

# 5. Stage files

git add .

# 6. Commit

git commit -m "Initial commit"

# 7. Connect GitHub repository

git remote add origin https://github.com/HabibHossain2024/REPOSITORY-NAME.git

# 8. Push

git push -u origin main
